OPTION 2: Use API Programmatically
──────────────────────────────────
from deployment_api import UHIPredictor

# Initialize
predictor = UHIPredictor(models_dir="models")

# Single prediction
features = [35.0, 0.3, 0.2, 0.15, 0.8, 26.0]  # Urban hotspot
result = predictor.predict_single(features)
print(f"Prediction: {result['predictions']['random_forest']['label']}")
print(f"Confidence: {result['predictions']['random_forest']['probability']:.2%}")

# Batch predictions
batch = [[35.0, 0.3, 0.2, 0.15, 0.8, 26.0],
         [25.0, 0.7, -0.2, 0.10, 0.2, 25.0]]
results = predictor.predict_batch(batch)

OPTION 3: Get Model Information
───────────────────────────────
info = predictor.get_model_info()
explanation = predictor.explain_prediction(features)
